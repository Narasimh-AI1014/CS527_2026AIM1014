% Sum of an array. Assume size is at address 0, elements from 0x4 onwards.
% Final result stored at address = size * 4 + 4.

x1 = 0       % index
x2 = 0       % sum
x3 = 0       % array base address
x4 = [x3]    % read size of array from address 0

.loopback
x15 = x1 - x4   % compare index with size
BEQ .exit       % if equal, exit loop

x3 = x3 + 4     % increase address
x6 = [x3]       % load array element
x2 = x2 + x6    % update sum
x1 = x1 + 1     % increment index
print x1        % print index to log
BAL .loopback   % branch always

.exit
[x3] = x2       % store sum right after the array
