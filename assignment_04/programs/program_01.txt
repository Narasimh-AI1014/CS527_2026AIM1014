% Count from 0 to 100000, printing every 1000 steps
x1 = 0          % counter
x2 = 1000       % print interval
x3 = 100000     % limit
x4 = 0          % temp for comparison

.loop
x1 = x1 + 1     % increment counter
x15 = x1 - x2   % check if multiple of 1000
BNE .skip       % if not, skip print
print x1        % print counter
x2 = x2 + 1000  % next print threshold
.skip
x15 = x1 - x3   % compare counter with limit
BLT .loop       % if counter < limit, continue

print x1        % final counter
