static Token match(TokenType type) {
    if (token_index < token_count && token_stream[token_index].type == type) {

           current_line_number, get_token_type_name(type), (token_index < token_count) ? token_stream[token_index].text : "EOF");
    exit(1);
}

static int peek(TokenType type) {

def create_project_zip(zip_name="simulator_project.zip"):
    print(f"[Packager] Generating separate workspace modules locally...")
    for filename, content in PROJECT_FILES.items():
        with open(filename, "w", encoding="utf-8") as f:
            f.write(content)
        print(f" -> Successfully written file: {filename}")

    print(f"\\n[Packager] Bundling modular workspace items into compressed archive: {zip_name}...")
    with zipfile.ZipFile(zip_name, "w", zipfile.ZIP_DEFLATED) as zipf:
        for filename in PROJECT_FILES.keys():
            if os.path.exists(filename):
                zipf.write(filename)
                print(f" -> Compressed and packed: {filename}")
    
    print(f"\\n[Packager] Success! Your clean multi-file lab directory is ready inside '{zip_name}'.")

if __name__ == "__main__":
    create_project_zip()
