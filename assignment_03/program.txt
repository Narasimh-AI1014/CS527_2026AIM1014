x1 = 0             % Initialize loop index index (k = 0)
x2 = [0]           % Load the array size N from data memory address 0
x3 = 4             % x3 = Base address of Array A (starts right after N)

% Calculate base address of Array B (4 + 4 * N)
x4 = x2 * 4
x4 = x4 + 4        % x4 now holds the starting address of Array B

% Calculate base address of Array C (4 + 8 * N)
x5 = x2 * 8
x5 = x5 + 4        % x5 now holds the destination address of Array C

.vector_loop
x6 = x1 - x2       % Compare current index with total size N
BEQ .done          % If index == N, exit loop

v1 = [x3]          % Parallel load 8 elements from Array A into vector register v1
v2 = [x4]          % Parallel load 8 elements from Array B into vector register v2

v3 = v1 + v2       % Execute vector addition across all 8 elements simultaneously

[x5] = v3          % Parallel store the 8 result elements into Array C memory

% Advance pointers by 32 bytes (8 elements * 4 bytes per integer)
x3 = x3 + 32
x4 = x4 + 32
x5 = x5 + 32

x1 = x1 + 8        % Increment loop index counter by 8 elements
BAL .vector_loop

.done
